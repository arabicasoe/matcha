function formatSize($bytes) {
    if ($bytes === false) return '-';
    $units = ['B', 'KB', 'MB', 'GB', 'TB'];
    $bytes = max($bytes, 0);
    $pow = floor(($bytes ? log($bytes) : 0) / log(1024));
    $pow = min($pow, count($units) - 1);
    $bytes /= pow(1024, $pow);
    return round($bytes, 2) . ' ' . $units[$pow];
}

function toBase64(str) { try { return btoa(unescape(encodeURIComponent(str))); } catch(e) { return btoa(str); } }

// FIXED UPLOAD DROPDOWN LOGIC
function toggleUploadDropdown(event) { 
    event.stopPropagation(); 
    const btn = event.currentTarget;
    const dropdown = document.getElementById('uploadDropdown');
    const rect = btn.getBoundingClientRect();
    const ddW = 220;
    let left = rect.left;
    if (left + ddW > window.innerWidth - 10) left = window.innerWidth - ddW - 10;
    if (left < 10) left = 10;
    dropdown.style.left = left + 'px';
    dropdown.style.top = (rect.bottom + 8) + 'px';
    dropdown.classList.toggle('show'); 
}

});

// HIDDEN INPUT HANDLERS FOR UPLOAD
function apiSingleUpload(file) {
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function(e) {
        const b64 = e.target.result.split(',')[1];
        const f = document.createElement('form');
        f.method = 'POST';

// === EDITOR BASE64 DECODER/ENCODER FIX ===
document.addEventListener('DOMContentLoaded', function() {
    const b64Input = document.getElementById('editor-b64');
    const visibleInput = document.getElementById('editor-visible');
    const editForm = document.getElementById('edit-form');

    // 1. Decode Base64 to readable text on load
    if (b64Input && visibleInput && b64Input.value) {
        try {
            visibleInput.value = decodeURIComponent(escape(window.atob(b64Input.value)));
        } catch (e) {
            visibleInput.value = "⚠️ Error: Failed to decode file content.";
        }
    }

    // 2. Encode text back to Base64 before submit
    if (editForm && b64Input && visibleInput) {
        editForm.addEventListener('submit', function(e) {
            try {
                b64Input.value = btoa(unescape(encodeURIComponent(visibleInput.value)));
            } catch (err) {
                e.preventDefault();
                alert("❌ Failed to save: Unsupported characters in file.");
            }
        });
    }
});
